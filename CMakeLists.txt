cmake_minimum_required(VERSION 3.22)
project(IA2A VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Configura -DJUCE_DIR con el directorio de JUCE 9.0.2")
endif()
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
get_directory_property(JUCE_VERSION DIRECTORY "${JUCE_DIR}" DEFINITION JUCE_VERSION)
if(NOT JUCE_VERSION VERSION_EQUAL "9.0.2")
    message(FATAL_ERROR "IA2A requiere JUCE 9.0.2")
endif()
juce_add_plugin(IA2A
    COMPANY_NAME "VitaPlug" BUNDLE_ID com.vitaplug.ia2a
    PLUGIN_MANUFACTURER_CODE Vtpl PLUGIN_CODE Ia2a PRODUCT_NAME IA2A
    FORMATS AU COPY_PLUGIN_AFTER_BUILD FALSE)
target_compile_definitions(IA2A PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(IA2A PRIVATE juce::juce_audio_utils)
include(dsp.cmake)
include(musical.cmake)
include(client.cmake)
juce_add_binary_data(IA2AArt SOURCES
    docs/ui/tres-frontales/assets/iaq8.png
    docs/ui/tres-frontales/assets/BarlowCondensed-SemiBold.ttf)
add_library(ia2a_session STATIC src/assist/Session.cpp src/audio/Capture.cpp src/audio/Compare.cpp)
target_link_libraries(ia2a_session PUBLIC ia2a_dsp)
target_include_directories(ia2a_session PUBLIC src)
target_link_libraries(IA2A PRIVATE ia2a_musical ia2a_client ia2a_session IA2AArt)
if(BUILD_TESTING)
    add_executable(SessionTest tests/assist/SessionTest.cpp)
    add_executable(AuxRouting tests/audio/Routing.cpp)
    add_executable(CompareTest tests/audio/CompareTest.cpp)
    foreach(test IN ITEMS SessionTest AuxRouting CompareTest)
        target_link_libraries(${test} PRIVATE ia2a_session)
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
endif()
target_sources(IA2A PRIVATE src/audio/PluginProcessor.cpp src/audio/State.cpp src/ui/Editor.cpp)
if(BUILD_TESTING)
    function(ia2a_plugin_test name file)
        add_executable(${name} "${file}")
        target_include_directories(${name} PRIVATE "${JUCE_DIR}/modules" src "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(${name} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
        target_link_libraries(${name} PRIVATE IA2A)
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    ia2a_plugin_test(EditorTest tests/ui/EditorTest.cpp)
    ia2a_plugin_test(ProcessorTest tests/e2e/ProcessorTest.cpp)
endif()
include(evaluation.cmake)
if(APPLE)
    add_custom_command(TARGET IA2A_AU POST_BUILD
        COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:IA2A_AU>"
        VERBATIM)
    if(BUILD_TESTING)
        add_executable(InstalledAU tests/audio/InstalledAU.cpp)
        target_link_libraries(InstalledAU PRIVATE "-framework AudioToolbox")
    endif()
endif()
